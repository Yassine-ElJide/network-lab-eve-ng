! R-CORE - inter-VLAN routing (router-on-a-stick) + OSPF
! Passwords removed for publication.
hostname R-CORE
!
! Sub-interfaces: one gateway per VLAN over the 802.1Q trunk
interface GigabitEthernet0/0
 no ip address
 no shutdown
!
interface GigabitEthernet0/0.10
 encapsulation dot1Q 10
 ip address 10.10.10.1 255.255.255.0
!
interface GigabitEthernet0/0.20
 encapsulation dot1Q 20
 ip address 10.10.20.1 255.255.255.0
 ip access-group USERS-TO-SERVERS in
!
interface GigabitEthernet0/0.99
 encapsulation dot1Q 99
 ip address 10.10.99.1 255.255.255.0
!
! Link to R-EDGE
interface GigabitEthernet0/1
 description To R-EDGE
 ip address 10.0.0.1 255.255.255.252
!
! OSPF
router ospf 1
 router-id 1.1.1.1
 passive-interface GigabitEthernet0/0.10
 passive-interface GigabitEthernet0/0.20
 passive-interface GigabitEthernet0/0.99
 network 10.10.10.0 0.0.0.255 area 0
 network 10.10.20.0 0.0.0.255 area 0
 network 10.10.99.0 0.0.0.255 area 0
 network 10.0.0.0 0.0.0.3 area 0
!
! Restrict Users (VLAN10) from reaching Server mgmt ports; allow web
ip access-list extended USERS-TO-SERVERS
 remark applied inbound on server sub-interface
 permit tcp 10.10.20.0 0.0.0.255 10.10.10.0 0.0.0.255 established
 deny   tcp 10.10.10.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 22
 deny   tcp 10.10.10.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 3389
 permit tcp 10.10.10.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 80
 permit tcp 10.10.10.0 0.0.0.255 10.10.20.0 0.0.0.255 eq 443
 deny   ip  10.10.10.0 0.0.0.255 10.10.20.0 0.0.0.255 log
 permit ip  any any
!
end
